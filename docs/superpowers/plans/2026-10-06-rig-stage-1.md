# Rig stage 1 - the mechanism without money - Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Everything that keeps a rented machine from outliving the side that took it - the rig journal and its
states, the lease, the reaper with its self-managing cron, the orphan rule, the secret filter, the ceilings, rig off
by default - working against doubles, before any machine is ever created.

**Architecture:** A new package `flotilla/rig/` beside `flotilla/lane/`, built the lane's way: an append-only JSON
Lines journal under a file lock (`core/storage.py`), folded into dataclasses with every field validated; a provider
behind a small interface, the first one calling the `vastai` CLI through an injected `run`; a reaper that is a pure
function of (journal, provider listing, clock); a cron module that edits only its own marked line. The CLI gains
`flotilla rig` (status), `rig renew` and `rig reap`. Creating machines, `rig open` and `rig run` are stage 2.

**Tech Stack:** Python 3.11+ standard library only; pytest; the `vastai` CLI (never called by the suite).

**Spec:** `docs/specs/2026-10-06-rig-design.md` (stage 1 of its section 11).

## Global Constraints

- English only in every file (`tools/check_no_cyrillic.py`); Python standard library only, no new dependency.
- Off by default: a fresh install has no key, no provider and no cron; `machine.toml` carries `rig = "off"` and
  `rig_provider = ""` until the person changes them.
- The provider's key lives only in `~/.config/flotilla/vast_api_key` (or `$XDG_CONFIG_HOME/flotilla/vast_api_key`),
  mode 600; wider permissions are a refusal. It is never in a project, a journal, an argument list, or printed.
- The person's ceilings: at most 1 machine at a time (`rig_max_machines`), at most 0.60 $/hour (`rig_max_hourly`),
  a session at most 8 hours (`rig_max_hours`). A malformed value falls back to the default, never above it.
- Lease 30 minutes; idle 15 minutes; reaper every 5 minutes; 3 failed drain attempts make a machine `stuck`;
  billing settling margin 2 minutes per machine.
- Every `vastai` call: `--raw`, `-y` where it asks, a 60 s timeout, stdin closed; a timeout, a non-zero exit or
  unparsable output is a failure, never a success.
- The journal keeps only id, GPU, price, state, address and label of an instance; provider output is scrubbed of
  every field named like key/token/secret/password before anything is printed or written.
- An instance without this machine's flotilla label is never touched.
- Nothing in the suite spends money, calls the real `vastai`, or touches the person's crontab.
- Tests: `uv run --with pytest python -m pytest tests -q` (the whole suite) stays green; a run's verdict is read from
  the tail of its output (`N passed`), never from a wrapper's exit code.

```python
"""The person's switch and ceilings for rented machines, read from `machine.toml` (rig design, sections 5 and 6).

Off unless the person wrote exactly `rig = "on"`: a fleet session cannot turn it on (the keys are the person's, kept
when the machine is measured again), and a value nobody meant - `yes`, `1`, a torn file - is off. A malformed ceiling
falls back to its default, never above it: a ceiling exists to stop spending, so doubt lowers it.
"""

from __future__ import annotations

import os
import re
import secrets as _secrets
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path

from flotilla.onboard.machine import read_machine

LABEL_PREFIX = "flotilla:"
OFF_LINE = ("rig is off on this machine: rented machines are used only after the person sets rig = \"on\" and "
            "rig_provider in machine.toml (README, section \"Rented machines (rig)\")")
DEFAULTS = {"rig_max_machines": 1, "rig_max_hourly": 0.60, "rig_max_hours": 8.0}
_KEY = re.compile(r"^[0-9a-f]{12}$")


@dataclass(frozen=True)
class RigSettings:
    on: bool
    provider: str
    max_machines: int = 1
    max_hourly: float = 0.60
    max_hours: float = 8.0


def _number(data: dict, key: str, *, whole: bool):
    value = data.get(key, DEFAULTS[key])
    if isinstance(value, bool) or not isinstance(value, (int, float)) or value != value or value <= 0:
        return DEFAULTS[key]
    if whole:
        return int(value) if float(value).is_integer() else DEFAULTS[key]
    return float(value)


def settings(state: Path) -> RigSettings:
    try:
        data = read_machine(state) or {}
    except (OSError, ValueError):
        data = {}
    provider = data.get("rig_provider")
    return RigSettings(on=data.get("rig") == "on", provider=provider if isinstance(provider, str) else "",
                       max_machines=_number(data, "rig_max_machines", whole=True),
                       max_hourly=_number(data, "rig_max_hourly", whole=False),
                       max_hours=_number(data, "rig_max_hours", whole=False))


def key_path(env: Mapping[str, str] = os.environ) -> Path:
    base = env.get("XDG_CONFIG_HOME") or str(Path.home() / ".config")
    return Path(base) / "flotilla" / "vast_api_key"


def machine_key(state: Path) -> str:
    """Which field machine made an instance: two machines on one provider account never reap each other's."""
    path = state / "rig" / "machine-key"
    try:
        found = path.read_text(encoding="utf-8").strip()
    except (OSError, UnicodeDecodeError):
        found = ""
    if _KEY.match(found):
        return found
    path.parent.mkdir(parents=True, exist_ok=True)
    made = _secrets.token_hex(6)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
    with os.fdopen(fd, "w", encoding="utf-8") as handle:
        handle.write(made + "\n")
    os.chmod(path, 0o600)
    return made


def label(key: str, machine_id: str) -> str:
    return f"{LABEL_PREFIX}{key}:{machine_id}"
```

```python
"""Provider output made safe to print and to journal (rig design, section 6).

A provider answers with more than it was asked: in the probe of 2026-10-06 `vastai create` returned the instance's
own API key, and it reached a log. Every answer passes here first: a field whose name says it holds a credential is
dropped at any depth, and its value is masked wherever else it appears - in another field, in stderr.
"""

from __future__ import annotations

import re

MASK = "[masked]"
_SECRET = re.compile(r"key|token|secret|password|passwd|credential", re.IGNORECASE)
_SHORTEST = 8   # a value this short is not a credential, and masking it would mask every "1" in the answer


def secret_name(name: str) -> bool:
    return bool(_SECRET.search(name))


def _collect(data, found: list[str]) -> None:
    if isinstance(data, dict):
        for name, value in data.items():
            if isinstance(name, str) and secret_name(name):
                if isinstance(value, str) and len(value) >= _SHORTEST:
                    found.append(value)
                elif isinstance(value, (dict, list)):
                    _strings(value, found)
            else:
                _collect(value, found)
    elif isinstance(data, list):
        for value in data:
            _collect(value, found)


def _strings(data, found: list[str]) -> None:
    if isinstance(data, str) and len(data) >= _SHORTEST:
        found.append(data)
    elif isinstance(data, dict):
        for value in data.values():
            _strings(value, found)
    elif isinstance(data, list):
        for value in data:
            _strings(value, found)


def scrub_text(text: str, values: list[str]) -> str:
    for value in sorted(set(values), key=len, reverse=True):
        text = text.replace(value, MASK)
    return text


def _clean(data, values: list[str]):
    if isinstance(data, dict):
        return {name: _clean(value, values) for name, value in data.items()
                if not (isinstance(name, str) and secret_name(name))}
    if isinstance(data, list):
        return [_clean(value, values) for value in data]
    if isinstance(data, str):
        return scrub_text(data, values)
    return data


def scrub(data) -> tuple[object, list[str]]:
    values: list[str] = []
    _collect(data, values)
    return _clean(data, values), values
```

```python
"""The rig's journal: sessions the person opened, machines rented in them, and every move of either (rig design,
sections 2 and 3).

The lane's way: an append-only event log under a file lock, folded into records, every field validated on the way
in, because every session may write here and a hand-written line must fold as unknown, never as a crash of the
reaper that keeps money from running. A move the state machine does not allow is refused, so the record can only
tell a story that happened: a machine is `gone` only after `draining`, and nothing comes back from `gone`.
"""

from __future__ import annotations

import datetime as dt
import math
from dataclasses import dataclass, fields as dc_fields

from flotilla.core.text import visible

KEY = "rig"
REQUESTED, PROVISIONING, READY, BUSY, DRAINING = "requested", "provisioning", "ready", "busy", "draining"
GONE, FAILED, STUCK = "gone", "failed", "stuck"
OPEN, CLOSING, CLOSED = "open", "closing", "closed"
MACHINE_STATES = (REQUESTED, PROVISIONING, READY, BUSY, DRAINING, GONE, FAILED, STUCK)
SESSION_STATES = (OPEN, CLOSING, CLOSED)
LIVE = tuple(state for state in MACHINE_STATES if state != GONE)   # whatever may still cost money
RENEWABLE = (REQUESTED, PROVISIONING, READY, BUSY)
MOVES = {
    REQUESTED: (PROVISIONING, FAILED, DRAINING),
    PROVISIONING: (READY, FAILED, DRAINING),
    READY: (BUSY, DRAINING),
    BUSY: (READY, DRAINING),
    DRAINING: (DRAINING, GONE, STUCK),
    STUCK: (STUCK, GONE),
    FAILED: (DRAINING, GONE),
    GONE: (),
}
SESSION_MOVES = {OPEN: (CLOSING,), CLOSING: (CLOSING, CLOSED), CLOSED: ()}
LEASE = dt.timedelta(minutes=30)
IDLE = dt.timedelta(minutes=15)
SETTLE = dt.timedelta(minutes=2)
ATTEMPTS = 3
MAX_TEXT = 300


class RigError(ValueError):
    """A move the rig's states do not allow, or a record that does not exist."""


@dataclass
class Session:
    id: str
    who: str = ""
    why: str = ""
    until: str = ""
    budget: float | None = None
    state: str = ""
    opened: str = ""
    ended: str = ""
    reason: str = ""


@dataclass
class Machine:
    id: str
    session: str = ""
    provider: str = ""
    instance: str = ""
    label: str = ""
    gpu: str = ""
    hourly: float | None = None
    state: str = ""
    created: str = ""
    lease_until: str = ""
    idle_since: str = ""
    attempts: int = 0
    reason: str = ""
    ended: str = ""


TEXT = ("who", "why", "session", "provider", "instance", "label", "gpu", "reason")
TIMES = ("until", "opened", "ended", "created", "lease_until", "idle_since")
MONEY = {"budget": 1e4, "hourly": 1e3}


def _aware(text) -> bool:
    try:
        return isinstance(text, str) and dt.datetime.fromisoformat(text).tzinfo is not None
    except ValueError:
        return False


def _clean(key, value):
    if key in TEXT:
        return visible(value[:MAX_TEXT]) if isinstance(value, str) else ""
    if key in TIMES:
        return value if value == "" or _aware(value) else ""
    if key in MONEY:
        ok = isinstance(value, (int, float)) and not isinstance(value, bool) and 0 <= value <= MONEY[key]
        return float(value) if ok else None
    if key == "attempts":
        return value if isinstance(value, int) and not isinstance(value, bool) and 0 <= value <= 1000 else 0
    return None


def _fold(records):
    sessions: dict[str, Session] = {}
    machines: dict[str, Machine] = {}
    for event in records:
        if not isinstance(event, dict) or not _aware(event.get("at")) or not isinstance(event.get("id"), str):
            continue
        kind, state = event.get("kind"), event.get("state")
        if kind == "session" and state in SESSION_STATES:
            item, names = sessions.get(event["id"]) or Session(id=event["id"]), Session
            found = sessions
        elif kind == "machine" and state in MACHINE_STATES:
            item, names = machines.get(event["id"]) or Machine(id=event["id"]), Machine
            found = machines
        else:
            continue   # a line flotilla never writes: skipped, never fatal
        for field in dc_fields(names):
            if field.name in ("id", "state") or field.name not in event:
                continue
            setattr(item, field.name, _clean(field.name, event[field.name]))
        item.state = state
        found[item.id] = item
    return sessions, machines


def _parse(text: str) -> dt.datetime | None:
    return dt.datetime.fromisoformat(text) if text else None


def _iso(moment: dt.datetime) -> str:
    return moment.isoformat(timespec="seconds")


class Rig:
    def __init__(self, store, *, clock=None):
        self.store = store
        self.clock = clock

    def now(self) -> dt.datetime:
        return self.clock() if self.clock else dt.datetime.now(dt.timezone.utc)

    def sessions(self) -> dict[str, Session]:
        return _fold(self.store.read(KEY).records)[0]

    def machines(self) -> dict[str, Machine]:
        return _fold(self.store.read(KEY).records)[1]

    def _append(self, tx, kind: str, item_id: str, state: str, **fields) -> None:
        tx.append({"at": _iso(self.now()), "kind": kind, "id": item_id, "state": state, **fields})

    def open_session(self, who: str, why: str, *, hours: float, budget: float) -> Session:
        with self.store.transaction(KEY) as tx:
            sessions, _ = _fold(tx.read().records)
            session_id = f"s{len(sessions) + 1}"
            now = self.now()
            self._append(tx, "session", session_id, OPEN, who=who, why=why, budget=budget, opened=_iso(now),
                         until=_iso(now + dt.timedelta(hours=hours)))
            return _fold(tx.read().records)[0][session_id]

    def set_session(self, session_id: str, state: str, reason: str = "", **fields) -> Session:
        with self.store.transaction(KEY) as tx:
            sessions, _ = _fold(tx.read().records)
            found = sessions.get(session_id)
            if found is None:
                raise RigError(f"no session {session_id}")
            if state not in SESSION_MOVES[found.state]:
                raise RigError(f"session {session_id} cannot move from {found.state} to {state}")
            extra = {"ended": _iso(self.now())} if state == CLOSED else {}
            self._append(tx, "session", session_id, state, **({"reason": reason} if reason else {}), **extra,
                         **fields)
            return _fold(tx.read().records)[0][session_id]

    def add_machine(self, session_id: str, provider: str, label_of) -> Machine:
        with self.store.transaction(KEY) as tx:
            sessions, machines = _fold(tx.read().records)
            if session_id not in sessions:
                raise RigError(f"no session {session_id}")
            machine_id = f"m{len(machines) + 1}"
            self._append(tx, "machine", machine_id, REQUESTED, session=session_id, provider=provider,
                         label=label_of(machine_id), lease_until=_iso(self.now() + LEASE))
            return _fold(tx.read().records)[1][machine_id]

    def move(self, machine_id: str, state: str, reason: str = "", **fields) -> Machine:
        with self.store.transaction(KEY) as tx:
            _, machines = _fold(tx.read().records)
            found = machines.get(machine_id)
            if found is None:
                raise RigError(f"no machine {machine_id}")
            if state not in MOVES[found.state]:
                raise RigError(f"machine {machine_id} cannot move from {found.state} to {state}")
            now = _iso(self.now())
            extra = {}
            if state == READY:
                extra["idle_since"] = now
            elif state == BUSY:
                extra["idle_since"] = ""
            if state == GONE:
                extra["ended"] = now
            self._append(tx, "machine", machine_id, state, **({"reason": reason} if reason else {}), **extra,
                         **fields)
            return _fold(tx.read().records)[1][machine_id]

    def renew(self, machine_ids=None) -> list[Machine]:
        renewed = []
        with self.store.transaction(KEY) as tx:
            _, machines = _fold(tx.read().records)
            until = _iso(self.now() + LEASE)
            for item in machines.values():
                if item.state in RENEWABLE and (machine_ids is None or item.id in machine_ids):
                    self._append(tx, "machine", item.id, item.state, lease_until=until)
                    renewed.append(item.id)
            if not renewed:
                return []
            machines = _fold(tx.read().records)[1]
            return [machines[machine_id] for machine_id in renewed]

    def spent(self, session_id: str, now: dt.datetime | None = None) -> float:
        """Money a session has cost by this side's count: each machine's price times the minutes begun since it was
        created, plus the minutes the provider keeps billing after a destroy (spec, section 5, layer 3)."""
        moment = now or self.now()
        total = 0.0
        for item in self.machines().values():
            if item.session != session_id or item.hourly is None or not item.created:
                continue
            end = _parse(item.ended) if item.state == GONE and item.ended else moment
            seconds = max(0.0, (end - _parse(item.created)).total_seconds())
            minutes = math.ceil(seconds / 60)
            if item.state == GONE:
                minutes += int(SETTLE.total_seconds() // 60)
            total += item.hourly * minutes / 60
        return total
```

```python
"""Where rented machines come from (rig design, section 2): an interface, and vast.ai through its `vastai` CLI.

Stage 1 needs only what the reaper uses - list the account's instances, destroy one. Every call is bounded: `--raw`
for JSON, `-y` where the CLI would ask (in the probe `destroy` without it hung on a prompt), stdin closed so a prompt
reads end of input, and a timeout. A timeout, a non-zero exit or output that does not parse is a failure: the reaper
reads "gone" only from a listing that lacks the instance, never from a command that seemed to succeed.

The key travels in the child's environment (`VAST_API_KEY`), never in its arguments, which every user of the
machine can read in the process table; and it is read from a file only its owner may read.
"""

from __future__ import annotations

import json
import os
import shutil
import stat
import subprocess
from dataclasses import dataclass
from pathlib import Path

from flotilla.rig.secrets import scrub, scrub_text
from flotilla.rig.settings import key_path


class ProviderError(RuntimeError):
    """The provider could not be asked, or its answer cannot be trusted."""


@dataclass(frozen=True)
class Listed:
    instance: str
    label: str
    status: str


class Vast:
    name = "vast"

    def __init__(self, *, key: Path, run=subprocess.run, which=shutil.which, timeout: float = 60.0,
                 home: Path | None = None):
        self.key = Path(key)
        self.run = run
        self.which = which
        self.timeout = timeout
        self.home = home or Path.home()

    def _secret(self) -> str:
        try:
            info = self.key.stat()
        except FileNotFoundError:
            raise ProviderError(f"no vast key at {self.key}; the person puts a scoped key there (mode 600)") from None
        if stat.S_IMODE(info.st_mode) & 0o077:
            raise ProviderError(f"{self.key} is readable by others (mode {stat.S_IMODE(info.st_mode):o}); "
                                f"`chmod 600 {self.key}` first")
        text = self.key.read_text(encoding="utf-8").strip()
        if not text:
            raise ProviderError(f"{self.key} is empty")
        return text

    def _exe(self) -> str:
        found = self.which("vastai")
        if found:
            return found
        local = self.home / ".local" / "bin" / "vastai"   # where pip --user puts it; cron's PATH lacks it
        if local.exists():
            return str(local)
        raise ProviderError("the vastai CLI is not installed (`pip install vastai`)")

    def _call(self, args: list[str], *, parse: bool = True):
        secret = self._secret()
        argv = [self._exe(), *args, "--raw"]
        env = {**os.environ, "VAST_API_KEY": secret}
        try:
            done = self.run(argv, capture_output=True, text=True, timeout=self.timeout, env=env,
                            stdin=subprocess.DEVNULL, check=False)
        except subprocess.TimeoutExpired:
            raise ProviderError(f"`vastai {' '.join(args)}` timed out after {self.timeout:g} s") from None
        except OSError as err:
            raise ProviderError(f"`vastai` could not be run: {err}") from None
        if done.returncode != 0:
            said = scrub_text((done.stderr or done.stdout or "").strip(), [secret])
            raise ProviderError(f"`vastai {' '.join(args)}` exited {done.returncode}: {said[:300]}")
        if not parse:
            return None
        try:
            data = json.loads(done.stdout)
        except json.JSONDecodeError:
            raise ProviderError(f"`vastai {' '.join(args)}` answered unparsable output") from None
        clean, _ = scrub(data)
        return clean

    def instances(self) -> list[Listed]:
        data = self._call(["show", "instances"])
        if not isinstance(data, list):
            raise ProviderError("`vastai show instances` answered something other than a list")
        found = []
        for row in data:
            if not isinstance(row, dict) or row.get("id") is None:
                continue
            label = row.get("label")
            status = row.get("actual_status") or row.get("cur_state") or ""
            found.append(Listed(str(row["id"]), label if isinstance(label, str) else "", str(status)))
        return found

    def destroy(self, instance: str) -> None:
        """Asks for the destroy; whether it happened is the listing's to say."""
        if not instance.isdigit():
            raise ProviderError(f"not a vast instance id: {instance!r}")
        self._call(["destroy", "instance", instance, "-y"], parse=False)


def provider_for(name: str, *, env=os.environ, run=subprocess.run) -> Vast:
    if name == "vast":
        return Vast(key=key_path(env), run=run)
    raise ProviderError(f"no provider `{name}`; this version knows `vast`")
```

```python
"""The reaper's line in the person's crontab (rig design, section 5, layer 1).

flotilla has no daemons; cron is what survives the session that rented a machine, and a reboot. The line is marked
with a comment, installed once, rewritten when the plugin moved (an update changes the versioned path it runs), and
removed when no machine and no session remain. Every other line of the crontab is kept as it was. A crontab that
could not be read is never written: replacing what we could not see would erase the person's jobs.
"""

from __future__ import annotations

import os
import shlex
import subprocess
from pathlib import Path

MARK = "# flotilla rig reaper"
CARRIED = ("PATH", "FLOTILLA_STATE_DIR", "XDG_CONFIG_HOME", "XDG_STATE_HOME")


class CronError(RuntimeError):
    """crontab could not be read or written."""


def reaper_line(*, entry: Path, python: str, state: Path, env=os.environ) -> str:
    carried = " ".join(f"{name}={shlex.quote(env[name])}" for name in CARRIED if env.get(name))
    log = shlex.quote(str(state / "rig" / "reap.log"))
    return (f"*/5 * * * * env {carried} {shlex.quote(python)} {shlex.quote(str(entry))} rig reap "
            f">> {log} 2>&1 {MARK}")


def entry_of(line: str) -> str:
    try:
        words = shlex.split(line.split(MARK)[0])
    except ValueError:
        return ""
    return words[words.index("rig") - 1] if "rig" in words and words.index("rig") > 0 else ""


def _read(run) -> list[str]:
    try:
        done = run(["crontab", "-l"], capture_output=True, text=True, timeout=30, check=False)
    except (OSError, subprocess.TimeoutExpired) as err:
        raise CronError(f"crontab could not be run: {err}") from None
    if done.returncode == 0:
        return done.stdout.splitlines()
    said = (done.stderr or done.stdout or "").strip()
    if "no crontab" in said.lower():
        return []
    raise CronError(f"crontab -l failed: {said[:200]}")


def _write(lines: list[str], run) -> None:
    text = "".join(f"{line}\n" for line in lines)
    try:
        done = run(["crontab", "-"], input=text, capture_output=True, text=True, timeout=30, check=False)
    except (OSError, subprocess.TimeoutExpired) as err:
        raise CronError(f"crontab could not be run: {err}") from None
    if done.returncode != 0:
        raise CronError(f"crontab - failed: {(done.stderr or done.stdout or '').strip()[:200]}")


def installed(*, run=subprocess.run) -> str:
    return next((line for line in _read(run) if line.endswith(MARK)), "")


def ensure(line: str, *, run=subprocess.run) -> str:
    lines = _read(run)
    ours = [item for item in lines if item.endswith(MARK)]
    if ours == [line]:
        return "unchanged"
    _write([item for item in lines if not item.endswith(MARK)] + [line], run)
    return "replaced" if ours else "installed"


def remove(*, run=subprocess.run) -> bool:
    lines = _read(run)
    kept = [item for item in lines if not item.endswith(MARK)]
    if len(kept) == len(lines):
        return False
    _write(kept, run)
    return True
```

```python
"""The reaper: what keeps a rented machine from outliving the side that took it (rig design, section 5, layer 1).

Run by cron every five minutes and by hand. In order:
1. a session past its end, or whose spending reached its budget, starts closing;
2. a machine whose lease expired, that sat ready for 15 minutes, or whose session is closing, starts draining;
3. every draining or stuck machine is destroyed and then looked for in the provider's listing - absent is `gone`,
   present or unanswered counts an attempt, and three attempts make it `stuck`, which stays loud until it clears;
4. an instance carrying this machine's flotilla label that the journal does not know as live is an orphan (a crash
   between create and the journal's write) and is destroyed; an instance without that label is never touched;
5. a closing session with nothing live closes, with what it cost.
"""

from __future__ import annotations

import datetime as dt
from dataclasses import dataclass, field

from flotilla.rig import journal as j
from flotilla.rig.provider import ProviderError
from flotilla.rig.settings import LABEL_PREFIX


@dataclass
class Outcome:
    lines: list[str] = field(default_factory=list)
    stuck: list[str] = field(default_factory=list)
    failed: bool = False


def _when(text: str) -> dt.datetime | None:
    return dt.datetime.fromisoformat(text) if text else None


def needs_reaper(rig: j.Rig) -> bool:
    return any(item.state in j.LIVE for item in rig.machines().values()) or \
        any(item.state in (j.OPEN, j.CLOSING) for item in rig.sessions().values())


def _close_sessions(rig: j.Rig, now: dt.datetime, out: Outcome) -> None:
    for session in rig.sessions().values():
        if session.state != j.OPEN:
            continue
        until = _when(session.until)
        reason = ""
        if until is not None and now >= until:
            reason = "session ended"
        elif session.budget is not None and rig.spent(session.id, now) >= session.budget:
            reason = "budget"
        if reason:
            rig.set_session(session.id, j.CLOSING, reason=reason)
            out.lines.append(f"session {session.id} closing: {reason}")


def _drain_reasons(rig: j.Rig, now: dt.datetime, out: Outcome) -> None:
    sessions = rig.sessions()
    for item in rig.machines().values():
        if item.state == j.FAILED and not item.instance:
            rig.move(item.id, j.GONE, reason=item.reason or "failed before an instance existed")
            out.lines.append(f"machine {item.id} gone: it never had an instance")
            continue
        if item.state not in (j.REQUESTED, j.PROVISIONING, j.READY, j.BUSY, j.FAILED):
            continue
        session = sessions.get(item.session)
        lease, idle = _when(item.lease_until), _when(item.idle_since)
        reason = ""
        if session is None or session.state != j.OPEN:
            reason = session.reason if session is not None and session.reason else "session ended"
        elif lease is None or now >= lease:
            reason = "lease expired"
        elif item.state == j.READY and idle is not None and now - idle >= j.IDLE:
            reason = "idle 15 min"
        elif item.state == j.FAILED:
            reason = item.reason or "failed"
        if reason:
            rig.move(item.id, j.DRAINING, reason=reason)
            out.lines.append(f"machine {item.id} draining: {reason}")


def _destroy_and_verify(rig: j.Rig, providers, out: Outcome, listings: dict) -> None:
    for item in rig.machines().values():
        if item.state not in (j.DRAINING, j.STUCK):
            continue
        try:
            provider = providers(item.provider)
            if item.instance:
                try:
                    provider.destroy(item.instance)
                except ProviderError as err:
                    out.lines.append(f"machine {item.id}: destroy failed: {err}")
            if item.provider not in listings:
                listings[item.provider] = provider.instances()
            listed = {entry.instance for entry in listings[item.provider]}
        except ProviderError as err:
            out.failed = True
            out.lines.append(f"machine {item.id}: the provider could not be asked: {err}")
            listed = None
        if listed is not None and item.instance not in listed:
            rig.move(item.id, j.GONE)
            out.lines.append(f"machine {item.id} gone (instance {item.instance or 'none'} no longer listed)")
            continue
        attempts = item.attempts + 1
        state = j.STUCK if attempts >= j.ATTEMPTS else j.DRAINING
        if item.state == j.STUCK:
            state = j.STUCK
        rig.move(item.id, state, attempts=attempts)
        if state == j.STUCK:
            out.stuck.append(item.id)
            out.lines.append(f"machine {item.id} STUCK after {attempts} attempts: instance {item.instance} may "
                             f"still cost money - check the provider's console")


def _orphans(rig: j.Rig, providers, machine_key: str, out: Outcome, listings: dict) -> None:
    prefix = f"{LABEL_PREFIX}{machine_key}:"
    machines = rig.machines()
    live_instances = {item.instance for item in machines.values() if item.state in j.LIVE and item.instance}
    names = {item.provider for item in machines.values()} | {"vast"}
    for name in sorted(names):
        try:
            provider = providers(name)
            listed = listings.get(name)
            if listed is None:
                listed = listings[name] = provider.instances()
        except ProviderError as err:
            if any(item.provider == name for item in machines.values()):
                out.failed = True
                out.lines.append(f"orphans on {name} not checked: {err}")
            continue
        for entry in listed:
            if not entry.label.startswith(prefix) or entry.instance in live_instances:
                continue
            try:
                provider.destroy(entry.instance)
                out.lines.append(f"orphan instance {entry.instance} ({entry.label}) destroyed")
            except ProviderError as err:
                out.failed = True
                out.lines.append(f"orphan instance {entry.instance} ({entry.label}): destroy failed: {err}")


def _finish_sessions(rig: j.Rig, now: dt.datetime, out: Outcome) -> None:
    machines = rig.machines().values()
    for session in rig.sessions().values():
        if session.state != j.CLOSING:
            continue
        if any(item.session == session.id and item.state in j.LIVE for item in machines):
            continue
        cost = rig.spent(session.id, now)
        rig.set_session(session.id, j.CLOSED)
        out.lines.append(f"session {session.id} closed: cost {cost:.2f} $")


def reap(rig: j.Rig, providers, *, machine_key: str) -> Outcome:
    out = Outcome()
    now = rig.now()
    _close_sessions(rig, now, out)
    _drain_reasons(rig, now, out)
    listings: dict = {}
    _destroy_and_verify(rig, providers, out, listings)
    listings.clear()   # destroys above changed what is listed: ask again before judging orphans
    _orphans(rig, providers, machine_key, out, listings)
    _finish_sessions(rig, now, out)
    return out
```

Note on `_orphans`: provider `vast` is always asked even with an empty journal, so an orphan left by a crash
before the first journal line is still found. When rig was never used, the reaper never runs (no cron line), so
this costs nothing on a machine without a key; a missing key on a machine with no rig machines prints nothing.

```python
from __future__ import annotations

import datetime as dt
import subprocess
import sys
from pathlib import Path

from flotilla.core import paths
from flotilla.core.storage import LocalLogStore, StorageCorrupt
from flotilla.rig import cron, provider, reaper
from flotilla.rig import journal as j
from flotilla.rig import settings as rs

CRON_RUN = subprocess.run
PROVIDERS = provider.provider_for
CLOCK = None
LOG_LIMIT = 1_000_000
ENTRY = Path(__file__).resolve().parents[2] / "bin" / "flotilla"


def _rig(state: Path) -> j.Rig:
    return j.Rig(LocalLogStore(state / "rig"), clock=CLOCK)


def _hhmm(text: str) -> str:
    return dt.datetime.fromisoformat(text).astimezone().strftime("%H:%M") if text else "?"


def _keep_log_small(state: Path) -> None:
    log = state / "rig" / "reap.log"
    try:
        if log.stat().st_size > LOG_LIMIT:
            log.write_text("", encoding="utf-8")
    except OSError:
        pass


def _cron_line(state: Path) -> str:
    return cron.reaper_line(entry=ENTRY, python=sys.executable, state=state)


def _keep_reaper(rig: j.Rig, state: Path) -> str:
    """Install, move or remove the reaper's line as the journal needs; "" or why it could not."""
    try:
        if reaper.needs_reaper(rig):
            cron.ensure(_cron_line(state), run=CRON_RUN)
        else:
            cron.remove(run=CRON_RUN)
    except cron.CronError as err:
        return str(err)
    return ""


def _status(state: Path, settings: rs.RigSettings) -> int:
    rig = _rig(state)
    machines = rig.machines()
    sessions = rig.sessions()
    live = [item for item in machines.values() if item.state in j.LIVE]
    if not settings.on and not live:
        print(rs.OFF_LINE)
        return 0
    word = "on" if settings.on else "OFF (machines below still live; the reaper drains them)"
    print(f"rig: {word}, provider {settings.provider or 'none'}; ceilings: {settings.max_machines} machine(s), "
          f"{settings.max_hourly:.2f} $/h, {settings.max_hours:g} h")
    for item in live:
        if item.state == j.STUCK:
            print(f"  STUCK machine {item.id}: instance {item.instance} may still cost money after "
                  f"{item.attempts} destroy attempts - check the provider's console")
    shown = [s for s in sessions.values() if s.state in (j.OPEN, j.CLOSING)]
    for session in shown:
        budget = f"{session.budget:.2f}" if session.budget is not None else "?"
        print(f"session {session.id} {session.state} ({session.who}: {session.why}) until {_hhmm(session.until)}, "
              f"spent {rig.spent(session.id):.2f} of {budget} $")
    if not shown:
        print("session: none open")
    for item in live:
        price = f"{item.hourly:.2f} $/h" if item.hourly is not None else "price unknown"
        print(f"  machine {item.id} {item.state}: instance {item.instance or 'none yet'}, {item.gpu or 'gpu ?'}, "
              f"{price}, lease until {_hhmm(item.lease_until)}"
              f"{f' ({item.reason})' if item.reason else ''}")
    try:
        line = cron.installed(run=CRON_RUN)
    except cron.CronError as err:
        print(f"reaper: crontab could not be read: {err}")
        return 0
    entry = cron.entry_of(line) if line else ""
    if not line:
        print("reaper: not installed" + (" - `flotilla rig renew` installs it" if live else ""))
    elif entry and not Path(entry).exists():
        print(f"reaper: its cron line runs {entry}, which no longer exists - `flotilla rig renew` rewrites it")
    else:
        print("reaper: installed (every 5 min)")
    return 0


def _renew(state: Path, settings: rs.RigSettings) -> int:
    rig = _rig(state)
    renewed = rig.renew()
    if renewed:
        print(f"renewed {len(renewed)} machine(s) until {_hhmm(renewed[0].lease_until)}")
    problem = _keep_reaper(rig, state)
    if problem:
        print(f"reaper: NOT installed: {problem} - a machine may outlive this session")
        return 1 if renewed else 0
    return 0


def _reap(state: Path) -> int:
    _keep_log_small(state)
    rig = _rig(state)
    out = reaper.reap(rig, PROVIDERS, machine_key=rs.machine_key(state))
    stamp = rig.now().isoformat(timespec="seconds")
    for line in out.lines:
        print(f"{stamp} {line}")
    problem = _keep_reaper(rig, state)
    if problem:
        print(f"{stamp} reaper line: {problem}")
    return 1 if out.stuck or out.failed else 0


def run_rig_command(args) -> int:
    state = paths.state_dir()
    settings = rs.settings(state)
    try:
        if args.action == "renew":
            return _renew(state, settings)
        if args.action == "reap":
            return _reap(state)
        return _status(state, settings)
    except StorageCorrupt as err:
        print(f"the rig journal is damaged: {err}")
        return 2
```
